cmake_minimum_required(VERSION 3.16)
project(display_link C)
add_library(display_link STATIC src/sdlp.c)
target_include_directories(display_link PUBLIC include)
target_compile_features(display_link PUBLIC c_std_11)
option(SDLP_BUILD_TESTS "Build portable protocol tests" OFF)
if(SDLP_BUILD_TESTS)
  enable_testing()
  add_executable(sdlp_test tests/sdlp_test.c)
  target_link_libraries(sdlp_test PRIVATE display_link)
  add_test(NAME sdlp COMMAND sdlp_test)
  add_executable(dlp_bandwidth_test tests/dlp_bandwidth_test.c)
  target_link_libraries(dlp_bandwidth_test PRIVATE display_link)
  add_test(NAME dlp_bandwidth COMMAND dlp_bandwidth_test)
  add_executable(dlp_window_test tests/dlp_window_test.c)
  target_link_libraries(dlp_window_test PRIVATE display_link)
  add_test(NAME dlp_window COMMAND dlp_window_test)
endif()
